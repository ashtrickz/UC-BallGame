using Cysharp.Threading.Tasks;

var gameplay = Game.Gameplay.Editor.GameplayDebugMenu.Find();
if (gameplay == null) throw new System.InvalidOperationException("Open gameplay before recording.");
Cysharp.Threading.Tasks.UniTask.Void(async () =>
{
    var rows = new System.Collections.Generic.List<object>();
    const string review = "Art/Reviews/ShotFeedback/2026-11-09/";
    var recorderObject = new UnityEngine.GameObject("Shot feedback review recorder");
    var recorder = recorderObject.AddComponent<Game.Gameplay.View.FrameRecorder>();
    recorder.Width = 540;
    recorder.Height = 1170;
    recorder.FrameRate = 30;
    try
    {
        foreach (var spec in new[] { new { name = "small", hold = 0.0 }, new { name = "large", hold = 2.0 } })
        {
            System.IO.File.WriteAllText(review + "recording-status.txt", "recording " + spec.name);
            gameplay.PlayWall(424242);
            var context = gameplay.Current;
            var model = context.Model;
            await Cysharp.Threading.Tasks.UniTask.WaitUntil(() => model.Phase == Game.Gameplay.Model.AttemptPhase.Waiting && !gameplay.IsBusy)
                .Timeout(System.TimeSpan.FromSeconds(15));
            gameplay.Stage.Input.SetVirtualPointer(false);
            await Cysharp.Threading.Tasks.UniTask.Yield();
            recorder.Begin("Temp/ShotFeedbackDemo/" + spec.name);
            await gameplay.Stage.Clock.Wait(0.5f, context.Token);
            var shot = Game.Integration.Game.GameplayAutoplay.HoldAsync(gameplay, model, spec.hold, context.Token);
            var previewFrame = -1;
            if (spec.hold > 0.5)
            {
                await Cysharp.Threading.Tasks.UniTask.WaitUntil(() => model.HoldSeconds >= spec.hold - 0.2)
                    .Timeout(System.TimeSpan.FromSeconds(12));
                previewFrame = System.Math.Max(0, recorder.Frames - 1);
            }
            await shot;
            var plan = model.Pending.Shot;
            await gameplay.Stage.Clock.Wait(0.23f, context.Token);
            var flightFrame = System.Math.Max(0, recorder.Frames - 1);
            await Cysharp.Threading.Tasks.UniTask.WaitUntil(() => model.Phase == Game.Gameplay.Model.AttemptPhase.Clearing)
                .Timeout(System.TimeSpan.FromSeconds(5));
            var impactFrame = System.Math.Max(0, recorder.Frames - 1);
            await gameplay.Stage.Clock.Wait(0.27f, context.Token);
            var peakFrame = System.Math.Max(0, recorder.Frames - 1);
            await gameplay.Stage.Clock.Wait(0.75f, context.Token);
            recorder.End();
            gameplay.Stage.Input.SetVirtualPointer(null);
            rows.Add(new { spec.name, spec.hold, frames = recorder.Frames, rate = recorder.FrameRate,
                previewFrame, flightFrame, impactFrame, peakFrame, radius = plan.BlastRadius, hits = plan.HitIds.Count });
            System.IO.File.WriteAllText(review + "demo-frames.json",
                Newtonsoft.Json.JsonConvert.SerializeObject(rows, Newtonsoft.Json.Formatting.Indented));
        }
        System.IO.File.WriteAllText(review + "recording-status.txt", "complete");
        UnityEngine.Debug.Log("[Shot feedback review] Recorded minimum and strong hits through the real controller.");
    }
    catch (System.Exception error)
    {
        System.IO.File.WriteAllText(review + "recording-status.txt", error.ToString());
        UnityEngine.Debug.LogException(error);
    }
    finally
    {
        recorder.End();
        gameplay.SetScriptedHold(null);
        gameplay.Stage.Input.SetVirtualPointer(null);
        UnityEngine.Object.Destroy(recorderObject);
    }
});
return "Recording started; status is in Art/Reviews/ShotFeedback/2026-11-09/recording-status.txt";
